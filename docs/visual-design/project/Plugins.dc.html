point-labels.js
export function pointLabels({ offset = 10 } = {}) {
  return (chart) => {
    chart.on('draw', (ctx) => {
      if (ctx.type !== 'point') return;
      ctx.group.elem('text', { x: ctx.x, y: ctx.y - offset },
        'ct-label').text(ctx.value.y);
    });
  };
}

new LineChart('#chart', data, { plugins: [pointLabels()] });
